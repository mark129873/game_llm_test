cmake_minimum_required(VERSION 3.24)
project(godot_qwen_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
option(QWEN_BUILD_GDEXTENSION "Build the Godot GDExtension wrapper" ON)
option(QWEN_BUILD_TESTS "Build core unit/smoke tests" ON)
option(QWEN_IOS_STATIC "Build static archives for the iOS XCFramework" OFF)
set(QWEN_OUTPUT_DIR "${ROOT_DIR}/bin" CACHE PATH "Godot native-library output directory")

if(NOT EXISTS "${ROOT_DIR}/vendor/llama.cpp/include/llama.h")
  message(FATAL_ERROR "Missing vendor/llama.cpp. Run scripts/fetch_dependencies.py first.")
endif()
# Deliberately in-process CPU-only inference. No HTTP, dlopen backends, or downloads.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMIT "a9d27ac693ff0a1d3cba449c912c2b891904a4c7" CACHE STRING "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT_DIR}/vendor/llama.cpp" "vendor/llama.cpp" EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_library(qwen_core STATIC src/qwen_core.cpp src/qwen_text.cpp)
target_include_directories(qwen_core PUBLIC include)
target_link_libraries(qwen_core PUBLIC llama Threads::Threads)

if(QWEN_BUILD_GDEXTENSION)
  if(NOT EXISTS "${ROOT_DIR}/vendor/godot-cpp/CMakeLists.txt")
    message(FATAL_ERROR "Missing vendor/godot-cpp. Run scripts/fetch_dependencies.py first.")
  endif()
  file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/godot_build_profile.json" CONTENT "{\"enabled_classes\":[\"RefCounted\",\"OS\"]}\n")
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_BINARY_DIR}/godot_build_profile.json" CACHE PATH "" FORCE)
  set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
  set(GODOTCPP_SYSTEM_HEADERS ON CACHE BOOL "" FORCE)
  add_subdirectory("${ROOT_DIR}/vendor/godot-cpp" "vendor/godot-cpp" EXCLUDE_FROM_ALL)
  if(QWEN_IOS_STATIC)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
      message(FATAL_ERROR "QWEN_IOS_STATIC requires CMAKE_SYSTEM_NAME=iOS and Apple Xcode SDKs.")
    endif()
    add_library(qwen_runner STATIC src/qwen_runner.cpp src/register_types.cpp)
  else()
    add_library(qwen_runner SHARED src/qwen_runner.cpp src/register_types.cpp)
  endif()
  target_link_libraries(qwen_runner PRIVATE qwen_core godot-cpp)
  target_include_directories(qwen_runner PRIVATE src)
  if(QWEN_IOS_STATIC)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/qwen_archives_$<CONFIG>.txt" CONTENT
      "$<TARGET_FILE:qwen_runner>\n$<TARGET_FILE:qwen_core>\n$<TARGET_FILE:godot-cpp>\n$<TARGET_FILE:llama>\n$<TARGET_FILE:ggml>\n$<TARGET_FILE:ggml-base>\n$<TARGET_FILE:ggml-cpu>\n")
  endif()
  if(WIN32)
    set(QWEN_PLATFORM windows)
    set_target_properties(qwen_runner PROPERTIES PREFIX "")
  elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(QWEN_PLATFORM ios)
  elseif(APPLE)
    set(QWEN_PLATFORM macos)
  else()
    set(QWEN_PLATFORM linux)
  endif()
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$" OR CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    set(QWEN_ARCH arm64)
  else()
    set(QWEN_ARCH x86_64)
  endif()
  set_target_properties(qwen_runner PROPERTIES
    OUTPUT_NAME "qwen_runner.${QWEN_PLATFORM}.${QWEN_ARCH}"
    LIBRARY_OUTPUT_DIRECTORY "${QWEN_OUTPUT_DIR}"
    RUNTIME_OUTPUT_DIRECTORY "${QWEN_OUTPUT_DIR}"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/archives")
  foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(qwen_runner PROPERTIES
      LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${QWEN_OUTPUT_DIR}"
      RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${QWEN_OUTPUT_DIR}")
  endforeach()
endif()

if(QWEN_BUILD_TESTS AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_testing()
  add_executable(qwen_text_test tests/text_test.cpp src/qwen_text.cpp)
  target_include_directories(qwen_text_test PRIVATE include)
  # Keep assertions active even in optimized verification builds.
  if(MSVC)
    target_compile_options(qwen_text_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(qwen_text_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME qwen_text COMMAND qwen_text_test)
  add_executable(qwen_error_test tests/error_test.cpp)
  target_link_libraries(qwen_error_test PRIVATE qwen_core)
  add_test(NAME qwen_error COMMAND qwen_error_test)
  add_executable(qwen_smoke tests/core_smoke.cpp)
  target_link_libraries(qwen_smoke PRIVATE qwen_core)
  add_executable(qwen_lifecycle tests/lifecycle_test.cpp)
  target_link_libraries(qwen_lifecycle PRIVATE qwen_core)
endif()
